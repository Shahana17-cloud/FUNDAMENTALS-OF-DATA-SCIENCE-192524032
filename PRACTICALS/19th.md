import numpy as np
from scipy import stats

drug = [10,12,11,9,13,12,14,10,11,13]
placebo = [5,6,4,7,5,6,8,4,5,7]

def confidence(data):
    mean = np.mean(data)
    sd = np.std(data, ddof=1)
    n = len(data)

    error = stats.t.ppf(0.975, n-1) * sd / np.sqrt(n)

    print("Mean:", mean)
    print("Lower Limit:", mean-error)
    print("Upper Limit:", mean+error)

print("Drug Group")
