import pandas as pd
import matplotlib.pyplot as plt
import scipy.stats as stats

age = [23,23,27,27,39,41,47,49,50,
       52,54,54,56,57,58,58,60,61]

fat = [9.5,26.5,7.8,17.8,31.4,25.9,27.4,27.2,31.2,
       34.6,42.5,28.8,33.4,30.2,34.1,32.9,41.2,35.7]

df = pd.DataFrame({'Age': age, 'Fat': fat})

print(df.mean())
print(df.median())
print(df.std())

df.boxplot()
plt.show()

plt.scatter(age, fat)
plt.xlabel("Age")
plt.ylabel("Body Fat")
plt.show()

stats.probplot(age, dist="norm", plot=plt)
plt.title("Q-Q Plot of Age")
plt.show()

stats.probplot(fat, dist="norm", plot=plt)
plt.title("Q-Q Plot of Body Fat")
plt.show()
