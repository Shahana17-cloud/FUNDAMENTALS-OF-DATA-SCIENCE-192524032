import pandas as pd
import matplotlib.pyplot as plt
import string

df = pd.read_csv("data.csv")

stop = ['the', 'is', 'and', 'a', 'an', 'to']

words = []

for review in df['feedback']:
    review = review.lower()

    for p in string.punctuation:
        review = review.replace(p, "")

    for word in review.split():
        if word not in stop:
            words.append(word)

freq = {}

for word in words:
    freq[word] = freq.get(word, 0) + 1

n = int(input("Enter N: "))

top = sorted(freq.items(), key=lambda x: x[1], reverse=True)[:n]

print(top)

x = [i[0] for i in top]
y = [i[1] for i in top]

plt.bar(x, y)
plt.xlabel("Words")
plt.ylabel("Frequency")
plt.title("Top Frequent Words")
plt.show()
