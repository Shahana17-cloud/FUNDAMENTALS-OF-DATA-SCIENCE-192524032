import pandas as pd

data = {'Review': ['good product', 'good quality', 'bad product']}

df = pd.DataFrame(data)

words = []

for review in df['Review']:
    words.extend(review.lower().split())

freq = {}

for word in words:
    if word in freq:
        freq[word] += 1
    else:
        freq[word] = 1

print(freq)
